local function Tween(obj, props, t, style, dir)
    TweenService:Create(obj,
        TweenInfo.new(t or 0.22,
            style or Enum.EasingStyle.Quart,
            dir   or Enum.EasingDirection.Out),
        props):Play()
end

local function IsAlive(p)
    local c = p.Character
    if not c then return false end
    local h = c:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end

local function GetPart(char, name)
    return char:FindFirstChild(name) or char:FindFirstChild("HumanoidRootPart")
end

local function W2V(pos)
    local v, vis = Camera:WorldToViewportPoint(pos)
    return Vector2.new(v.X, v.Y), v.Z > 0 and vis
end

local ScreenCenter = function()
    return Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
end

-- ============================================================
-- HITBOX EXPANDER — resize Head part client-side
-- ============================================================
local HitboxConnections = {}

local function ApplyHitbox(player)
    if player == LocalPlayer then return end
    local function onChar(char)
        char:WaitForChild(Cfg.HitboxPart, 5)
        local part = char:FindFirstChild(Cfg.HitboxPart)
        if not part then return end
        local conn = RunService.Heartbeat:Connect(function()
            if not Cfg.HitboxEnabled then return end
            part.Size = Vector3.new(Cfg.HitboxSize, Cfg.HitboxSize, Cfg.HitboxSize)
            part.Transparency = Cfg.HitboxVisible and 0.5 or 1
            part.LocalTransparencyModifier = Cfg.HitboxVisible and 0.5 or 1
        end)
        if HitboxConnections[player] then
            HitboxConnections[player]:Disconnect()
        end
        HitboxConnections[player] = conn
    end
    if player.Character then onChar(player.Character) end
    player.CharacterAdded:Connect(onChar)
end

local function RemoveHitbox(player)
    if HitboxConnections[player] then
        HitboxConnections[player]:Disconnect()
        HitboxConnections[player] = nil
    end
    -- restore size
    if player.Character then
        local part = player.Character:FindFirstChild(Cfg.HitboxPart)
        if part then
            part.Size = Vector3.new(2, 2, 2) -- roblox head default
        end
    end
end

for _, p in ipairs(Players:GetPlayers()) do ApplyHitbox(p) end
Players.PlayerAdded:Connect(ApplyHitbox)
Players.PlayerRemoving:Connect(function(p)
    RemoveHitbox(p)
    HitboxConnections[p] = nil
end)

-- ============================================================
-- SHOOT-DETECT HEADSHOT BOX
-- ============================================================
-- When a player fires toward our character, draw a highlighted
-- box around their Head in screen space.  Clicking inside it =
-- headshot.  Box size controlled by ShootBoxSize slider.
-- ============================================================
local ShootBoxDrawings = {}  -- [player] = { box, outline, timer }

local function SpawnShootBox(player)
    if ShootBoxDrawings[player] then return end

    local outline = Drawing.new("Square")
    outline.Thickness  = 3
    outline.Filled     = false
    outline.Color      = Cfg.ShootBoxOutline
    outline.Visible    = false
    outline.ZIndex     = 10

    local box = Drawing.new("Square")
    box.Thickness  = 1.5
    box.Filled     = false
    box.Color      = Cfg.ShootBoxColor
    box.Visible    = false
    box.ZIndex     = 11

    local label = Drawing.new("Text")
    label.Text    = "HEADSHOT"
    label.Size    = 11
    label.Font    = Drawing.Fonts.UI
    label.Outline = true
    label.Color   = Cfg.ShootBoxOutline
    label.Center  = true
    label.Visible = false
    label.ZIndex  = 12

    ShootBoxDrawings[player] = {
        box     = box,
        outline = outline,
        label   = label,
        alpha   = 0,
        active  = false,
        timer   = 0,
    }
end

local function TriggerShootBox(player)
    local d = ShootBoxDrawings[player]
    if not d then return end
    d.active = true
    d.timer  = Cfg.ShootBoxFade
    d.alpha  = 1
end

-- Detect incoming fire: watch for tool activation / remotes
-- Da Hood uses RemoteEvents for shooting — we hook tool activation
-- as a lightweight passive-only signal
local function WatchShooting(player)
    if player == LocalPlayer then return end
    local function onChar(char)
        char.ChildAdded:Connect(function(child)
            if child:IsA("Tool") then
                child.Activated:Connect(function()
                    -- player fired their tool — trigger box
                    if Cfg.ShootBoxEnabled and IsAlive(player) then
                        TriggerShootBox(player)
                    end
                end)
            end
        end)
        -- also catch already-equipped tools
        for _, child in ipairs(char:GetChildren()) do
            if child:IsA("Tool") then
                child.Activated:Connect(function()
                    if Cfg.ShootBoxEnabled and IsAlive(player) then
                        TriggerShootBox(player)
                    end
                end)
            end
        end
    end
    if player.Character then onChar(player.Character) end
    player.CharacterAdded:Connect(onChar)
    SpawnShootBox(player)
end

for _, p in ipairs(Players:GetPlayers()) do WatchShooting(p) end
Players.PlayerAdded:Connect(function(p) task.wait(1); WatchShooting(p) end)
Players.PlayerRemoving:Connect(function(p)
    local d = ShootBoxDrawings[p]
    if d then
        d.box:Remove(); d.outline:Remove(); d.label:Remove()
        ShootBoxDrawings[p] = nil
    end
end)

-- Update shoot boxes in RenderStepped
RunService.RenderStepped:Connect(function(dt)
    for player, d in pairs(ShootBoxDrawings) do
        local visible = Cfg.ShootBoxEnabled and d.active and IsAlive(player)

        if not visible then
            d.box.Visible     = false
            d.outline.Visible = false
            d.label.Visible   = false
            if d.timer <= 0 then d.active = false end
            d.timer = math.max(0, d.timer - dt)
            continue
        end

        local char = player.Character
        local head = char and char:FindFirstChild("Head")
        if not head then continue end

        local screenPos, onScreen = W2V(head.Position)
        if not onScreen then
            d.box.Visible = false; d.outline.Visible = false; d.label.Visible = false
            d.timer = math.max(0, d.timer - dt)
            if d.timer <= 0 then d.active = false end
            continue
        end

        -- fade out
        d.timer = math.max(0, d.timer - dt)
        d.alpha = d.timer / Cfg.ShootBoxFade

        local sz     = Cfg.ShootBoxSize * 6  -- pixel size from stud config
        local half   = sz / 2
        local pos    = Vector2.new(screenPos.X - half, screenPos.Y - half)
        local size   = Vector2.new(sz, sz)

        -- Outer glow outline
        d.outline.Position    = Vector2.new(pos.X - 2, pos.Y - 2)
        d.outline.Size        = Vector2.new(sz + 4, sz + 4)
        d.outline.Transparency = 1 - d.alpha * 0.9
        d.outline.Visible     = true

        -- Inner box
        d.box.Position        = pos
        d.box.Size            = size
        d.box.Transparency    = 1 - d.alpha
        d.box.Visible         = true

        -- Label
        d.label.Position      = Vector2.new(screenPos.X, pos.Y - 16)
        d.label.Transparency  = 1 - d.alpha
        d.label.Visible       = true

        if d.timer <= 0 then d.active = false end
    end
end)

-- ============================================================
-- AIMBOT — FIXED
-- Core fix: use Camera.CFrame manipulation inside RenderStepped
-- with a proper lerp that converges. Previous version used
-- Camera:WorldToViewportPoint without locking the CFrame first.
-- This version sets Camera.CFrame directly with slerp on the
-- rotational component only — position never moves.
-- ============================================================
local FOVCircle = Drawing.new("Circle")
FOVCircle.Thickness    = 1.5
FOVCircle.Color        = Cfg.Accent
FOVCircle.Filled       = false
FOVCircle.Transparency = 0.75
FOVCircle.Visible      = false

local function GetTarget()
    local best     = nil
    local bestDist = Cfg.AimbotFOV
    local center   = ScreenCenter()

    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        if not IsAlive(p) then continue end
        if Cfg.TeamCheck and p.Team == LocalPlayer.Team then continue end

        local char = p.Character
        local bone = GetPart(char, Cfg.AimbotBone)
        if not bone then continue end

        if Cfg.VisCheck then
            local origin = Camera.CFrame.Position
            local dir    = (bone.Position - origin).Unit * 999
            local ray    = RaycastParams.new()
            ray.FilterType = Enum.RaycastFilterType.Blacklist
            ray.FilterDescendantsInstances = {
                LocalPlayer.Character or {},
                workspace:FindFirstChild("Ignore") or Instance.new("Folder"),
            }
            local result = workspace:Raycast(origin, dir, ray)
            if result and not char:IsAncestorOf(result.Instance) then continue end
        end

        local screenPos, onScreen = W2V(bone.Position)
        if not onScreen then continue end

        local dist = (center - screenPos).Magnitude
        if dist < bestDist then
            bestDist = dist
            best     = p
        end
    end
    return best
end

-- THE FIX: RenderStepped + CFrame slerp + prediction offset
RunService.RenderStepped:Connect(function(dt)
    -- FOV circle
    FOVCircle.Position = ScreenCenter()
    FOVCircle.Radius   = Cfg.AimbotFOV
    FOVCircle.Visible  = Cfg.ShowFOV and Cfg.AimbotEnabled

    if not Cfg.AimbotEnabled then return end
    if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return end

    local target = GetTarget()
    if not target or not target.Character then return end

    local bone = GetPart(target.Character, Cfg.AimbotBone)
    if not bone then return end

    local aimPos = bone.Position

    -- Movement prediction: offset by velocity * strength
    if Cfg.PredictMovement then
        local root = target.Character:FindFirstChild("HumanoidRootPart")
        if root then
            local vel = root.AssemblyLinearVelocity
            aimPos = aimPos + vel * Cfg.PredictionStrength
        end
    end

    -- Build target CFrame (camera position unchanged, only rotation)
    local targetCF = CFrame.new(Camera.CFrame.Position, aimPos)

    -- Slerp rotation only — this is the fix
    local smooth   = math.clamp(Cfg.AimbotSmoothing, 0.01, 1.0)
    Camera.CFrame  = Camera.CFrame:Lerp(targetCF, smooth)
end)

local ESPStore = {}

local function MakeESP(player)
    ESPStore[player] = {
        box      = Drawing.new("Square"),
        outline  = Drawing.new("Square"),
        name     = Drawing.new("Text"),
        dist     = Drawing.new("Text"),
        hpBg     = Drawing.new("Square"),
        hp       = Drawing.new("Square"),
        tracer   = Drawing.new("Line"),
        snap     = Drawing.new("Line"),
    }
    local e = ESPStore[player]
    e.box.Filled = false;    e.box.Thickness = 1.5
    e.outline.Filled = false; e.outline.Thickness = 3
    e.name.Font = Drawing.Fonts.UI;  e.name.Size = 13; e.name.Outline = true; e.name.Center = true
    e.dist.Font = Drawing.Fonts.UI;  e.dist.Size = 11; e.dist.Outline = true; e.dist.Center = true
    e.hpBg.Filled = true;    e.hpBg.Color = Color3.fromRGB(15,15,15); e.hpBg.Transparency = 0.4
    e.hp.Filled = true
    e.tracer.Thickness = 1
    e.snap.Thickness = 1; e.snap.Color = Color3.fromRGB(255,255,100)
end

local function KillESP(player)
    if ESPStore[player] then
        for _, d in pairs(ESPStore[player]) do
            d.Visible = false; d:Remove()
        end
        ESPStore[player] = nil
    end
end

local function UpdateESP()
    for player, e in pairs(ESPStore) do
        local show = Cfg.ESPEnabled and IsAlive(player)

        if not show then
            for _, d in pairs(e) do d.Visible = false end
            continue
        end

        local char = player.Character
        local root = char:FindFirstChild("HumanoidRootPart")
        local head = char:FindFirstChild("Head")
        local hum  = char:FindFirstChildOfClass("Humanoid")
        if not (root and head and hum) then continue end

        local rPos, onScreen = W2V(root.Position)
        local hPos           = W2V(head.Position + Vector3.new(0, 0.65, 0))
        if not onScreen then
            for _, d in pairs(e) do d.Visible = false end
            continue
        end

        local boxH  = math.abs(rPos.Y - hPos.Y) * 2.3
        local boxW  = boxH * 0.55
        local boxX  = rPos.X - boxW / 2
        local boxY  = hPos.Y - boxH * 0.06
        local dist  = math.floor((Camera.CFrame.Position - root.Position).Magnitude)
        local hpPct = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
        local col   = Cfg.ESPTeamColor and (player.Team and player.TeamColor.Color or Cfg.Accent) or Cfg.Accent

        -- Outline glow
        e.outline.Visible  = Cfg.ESPBoxes
        e.outline.Position = Vector2.new(boxX - 1, boxY - 1)
        e.outline.Size     = Vector2.new(boxW + 2, boxH + 2)
        e.outline.Color    = Color3.fromRGB(0, 0, 0)
        e.outline.Transparency = 0.6

        -- Box
        e.box.Visible  = Cfg.ESPBoxes
        e.box.Position = Vector2.new(boxX, boxY)
        e.box.Size     = Vector2.new(boxW, boxH)
        e.box.Color    = col

        -- Name
        e.name.Visible   = Cfg.ESPNames
        e.name.Position  = Vector2.new(rPos.X, boxY - 17)
        e.name.Text      = player.DisplayName
        e.name.Color     = Cfg.TextCol

        -- Distance
        e.dist.Visible   = Cfg.ESPDistance
        e.dist.Position  = Vector2.new(rPos.X, boxY + boxH + 3)
        e.dist.Text      = dist .. "m"
        e.dist.Color     = Cfg.SubText

        -- Health bar (left side)
        local barX = boxX - 5
        e.hpBg.Visible  = Cfg.ESPHealth
        e.hpBg.Position = Vector2.new(barX, boxY)
        e.hpBg.Size     = Vector2.new(3, boxH)

        e.hp.Visible  = Cfg.ESPHealth
        e.hp.Position = Vector2.new(barX, boxY + boxH * (1 - hpPct))
        e.hp.Size     = Vector2.new(3, boxH * hpPct)
        e.hp.Color    = Color3.fromRGB(
            math.floor(255 * (1 - hpPct)),
            math.floor(255 * hpPct),
            50
        )

        -- Tracer from bottom center
        e.tracer.Visible = Cfg.ESPTracers
        e.tracer.From    = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
        e.tracer.To      = Vector2.new(rPos.X, rPos.Y)
        e.tracer.Color   = col

        -- Snapline to feet
        e.snap.Visible = Cfg.ESPSnapline
        e.snap.From    = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        e.snap.To      = Vector2.new(rPos.X, boxY + boxH)
        e.snap.Color   = Color3.fromRGB(255,255,100)
    end
end

for _, p in ipairs(Players:GetPlayers()) do

end

Players.PlayerRemoving:Connect(KillESP)

local function Slider(page, label, key, min, max, order, isFloat, callback)
    local val = Cfg[key]

    local row = Instance.new("Frame")
    row.Size  = UDim2.new(1, 0, 0, 54)
    row.BackgroundColor3 = Cfg.LightBG
    row.BorderSizePixel = 0
    row.LayoutOrder = order
    row.ZIndex = 5
    row.Parent = page
    Instance.new("UICorner", row).CornerRadius = UDim.new(0, 8)

    local lbl = Instance.new("TextLabel")
    lbl.Size  = UDim2.new(0.72, 0, 0, 24)
    lbl.Position = UDim2.new(0, 12, 0, 6)
    lbl.BackgroundTransparency = 1
    lbl.Text  = label
    lbl.Font  = Enum.Font.Gotham
    lbl.TextSize = 13
    lbl.TextColor3 = Cfg.TextCol
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.ZIndex = 6
    lbl.Parent = row

    local function fmt(v) return isFloat and string.format("%.2f", v) or tostring(v) end

    local valLbl = Instance.new("TextLabel")
    valLbl.Size  = UDim2.new(0.28, -12, 0, 24)
    valLbl.Position = UDim2.new(0.72, 0, 0, 6)
    valLbl.BackgroundTransparency = 1
    valLbl.Text  = fmt(val)
    valLbl.Font  = Enum.Font.GothamBold
    valLbl.TextSize = 13
    valLbl.TextColor3 = Cfg.Accent
    valLbl.TextXAlignment = Enum.TextXAlignment.Right
    valLbl.ZIndex = 6
    valLbl.Parent = row

    local track = Instance.new("Frame")
    track.Size  = UDim2.new(1, -24, 0, 4)
    track.Position = UDim2.new(0, 12, 0, 38)
    track.BackgroundColor3 = Cfg.MidBG
    track.BorderSizePixel = 0
    track.ZIndex = 6
    track.Parent = row
    Instance.new("UICorner", track).CornerRadius = UDim.new(1,0)

    local pct = (val - min) / (max - min)

    local fill = Instance.new("Frame")
    fill.Size  = UDim2.new(pct, 0, 1, 0)
    fill.BackgroundColor3 = Cfg.Primary
    fill.BorderSizePixel = 0
    fill.ZIndex = 7
    fill.Parent = track
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1,0)
    local fGrad = Instance.new("UIGradient")
    fGrad.Color = ColorSequence.new(Cfg.Primary, Cfg.Accent)
    fGrad.Parent = fill

    local knob = Instance.new("Frame")
    knob.Size  = UDim2.new(0, 12, 0, 12)
    knob.AnchorPoint = Vector2.new(0.5,0.5)
    knob.Position = UDim2.new(pct, 0, 0.5, 0)
    knob.BackgroundColor3 = Color3.new(1,1,1)
    knob.BorderSizePixel = 0
    knob.ZIndex = 8
    knob.Parent = track
    Instance.new("UICorner", knob).CornerRadius = UDim.new(1,0)

    local dragging = false

    local function applyDrag(input)
    local rel = math.clamp(
        (input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X,
        0, 1
    )
    local newVal
    if isFloat then
        newVal = math.floor((min + (max - min) * rel) * 100 + 0.5) / 100
    else
        newVal = math.floor(min + (max - min) * rel + 0.5)
    end
    Cfg[key]      = newVal
    valLbl.Text   = fmt(newVal)
    fill.Size     = UDim2.new(rel, 0, 1, 0)
    knob.Position = UDim2.new(rel, 0, 0.5, 0)
    if callback then callback(newVal) end
end

    track.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true; applyDrag(i)
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and i.UserInputType == Enum.UserInputType.MouseMovement then
            applyDrag(i)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    end)
end

    old.ico.TextColor3 = Cfg.SubText

    cur.ico.TextColor3 = Cfg.Accent
    cur.lbl.TextColor3 = Cfg.TextCol
    Pages[name].Visible = true
end
